=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const typedName = ref('')

const matches = computed(() => typedName.value === props.user.name)

watch(() => props.open, (isOpen) => {
  if (!isOpen) typedName.value = ''
})

function onCancel() {
  emit('update:open', false)
}

function onConfirm() {
  if (!matches.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          You are about to permanently delete <strong class="text-highlighted">{{ user.name }}</strong>.
          This action cannot be undone. To confirm, type the user's name below.
        </p>
        <UInput
          v-model="typedName"
          :placeholder="user.name"
          class="w-full"
          autocomplete="off"
          @keydown.enter="onConfirm"
        />
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          label="Cancel"
          color="neutral"
          variant="outline"
          @click="onCancel"
        />
        <UButton
          label="Delete"
          color="error"
          :disabled="!matches"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
