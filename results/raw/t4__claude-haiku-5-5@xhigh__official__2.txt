import { db, schema } from 'hub:db'
import { count, eq } from 'drizzle-orm'

export default defineEventHandler(async () => {
  // Member count per project (subquery)
  const memberCounts = db
    .select({
      projectId: schema.members.projectId,
      memberCount: count(schema.members.id).as('member_count'),
    })
    .from(schema.members)
    .groupBy(schema.members.projectId)
    .as('member_counts')

  // Task count per project (subquery)
  const taskCounts = db
    .select({
      projectId: schema.tasks.projectId,
      taskCount: count(schema.tasks.id).as('task_count'),
    })
    .from(schema.tasks)
    .groupBy(schema.tasks.projectId)
    .as('task_counts')

  // Single statement: every project, LEFT JOINed so projects with no members/tasks are kept
  const rows = await db
    .select({
      id: schema.projects.id,
      name: schema.projects.name,
      memberCount: memberCounts.memberCount,
      taskCount: taskCounts.taskCount,
    })
    .from(schema.projects)
    .leftJoin(memberCounts, eq(memberCounts.projectId, schema.projects.id))
    .leftJoin(taskCounts, eq(taskCounts.projectId, schema.projects.id))
    .orderBy(schema.projects.id)

  return rows.map(row => ({
    id: row.id,
    name: row.name,
    memberCount: row.memberCount ?? 0,
    taskCount: row.taskCount ?? 0,
  }))
})
