=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: [id: string]
}>()

const confirmation = ref('')

const canDelete = computed(() => confirmation.value === props.user.name)

watch(
  () => props.open,
  (isOpen) => {
    if (!isOpen) confirmation.value = ''
  }
)

watch(
  () => props.user.id,
  () => {
    confirmation.value = ''
  }
)

function close() {
  emit('update:open', false)
}

function onConfirm() {
  if (!canDelete.value) return
  emit('confirm', props.user.id)
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl sm:max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. This will permanently delete the user
          <span class="font-semibold text-highlighted">{{ user.name }}</span>
          and remove all of their associated data.
        </p>
        <div class="space-y-2">
          <label for="delete-user-confirm" class="block text-sm">
            Type <span class="font-mono font-semibold">{{ user.name }}</span> to confirm.
          </label>
          <UInput
            id="delete-user-confirm"
            v-model="confirmation"
            autocomplete="off"
            class="w-full"
          />
        </div>
      </div>
    </template>
    <template #footer>
      <div class="flex justify-end gap-2 w-full">
        <UButton color="neutral" variant="outline" @click="close">Cancel</UButton>
        <UButton color="error" :disabled="!canDelete" @click="onConfirm">Delete</UButton>
      </div>
    </template>
  </UModal>
</template>
