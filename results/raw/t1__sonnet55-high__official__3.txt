=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string, name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const confirmation = ref('')

const matches = computed(() => confirmation.value === props.user.name)

watch(
  () => props.open,
  (value) => {
    if (!value) {
      confirmation.value = ''
    }
  }
)

function onUpdateOpen(value: boolean) {
  emit('update:open', value)
}

function onCancel() {
  emit('update:open', false)
}

function onConfirm() {
  if (!matches.value) {
    return
  }
  emit('confirm')
  emit('update:open', false)
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="onUpdateOpen"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          You are about to permanently delete
          <strong class="text-highlighted">{{ props.user.name }}</strong>.
          This action cannot be undone and all data associated with this user will be lost.
        </p>

        <UFormField
          :label="`Type &quot;${props.user.name}&quot; to confirm`"
          required
        >
          <UInput
            v-model="confirmation"
            class="w-full"
            :placeholder="props.user.name"
            autocomplete="off"
            @keydown.enter="onConfirm"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          color="neutral"
          variant="outline"
          label="Cancel"
          @click="onCancel"
        />
        <UButton
          color="error"
          label="Delete"
          :disabled="!matches"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
