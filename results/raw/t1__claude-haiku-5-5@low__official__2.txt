=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: []
}>()

const typedName = ref('')

const matches = computed(() => typedName.value === props.user.name)

watch(
  () => props.open,
  (isOpen) => {
    if (isOpen) typedName.value = ''
  }
)

function onConfirm() {
  if (!matches.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl sm:max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. The user <strong>{{ user.name }}</strong> and all of their associated data will be permanently deleted.
        </p>

        <div class="space-y-2">
          <label for="delete-user-confirm-input" class="block text-sm">
            Type <strong>{{ user.name }}</strong> to confirm.
          </label>
          <UInput
            id="delete-user-confirm-input"
            v-model="typedName"
            class="w-full"
            :placeholder="user.name"
            autocomplete="off"
          />
        </div>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="emit('update:open', false)">
          Cancel
        </UButton>
        <UButton color="error" :disabled="!matches" @click="onConfirm">
          Delete
        </UButton>
      </div>
    </template>
  </UModal>
</template>
=== END ===
