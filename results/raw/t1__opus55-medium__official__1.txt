=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': [user: { id: string; name: string }]
}>()

const confirmation = ref('')

const isMatch = computed(() => confirmation.value === props.user.name)

watch(() => props.open, (value) => {
  if (!value) {
    confirmation.value = ''
  }
})

watch(() => props.user.id, () => {
  confirmation.value = ''
})

function setOpen(value: boolean) {
  emit('update:open', value)
}

function onCancel() {
  setOpen(false)
}

function onConfirm() {
  if (!isMatch.value) return
  emit('confirm', props.user)
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="setOpen"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          You are about to permanently delete the user
          <span class="font-semibold text-highlighted">{{ props.user.name }}</span>.
          This action cannot be undone. To confirm, type the user's name below.
        </p>

        <UFormField :label="`Type &quot;${props.user.name}&quot; to confirm`">
          <UInput
            v-model="confirmation"
            :placeholder="props.user.name"
            autocomplete="off"
            class="w-full"
            @keydown.enter.prevent="onConfirm"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          label="Cancel"
          color="neutral"
          variant="outline"
          @click="onCancel"
        />
        <UButton
          label="Delete"
          color="error"
          :disabled="!isMatch"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
