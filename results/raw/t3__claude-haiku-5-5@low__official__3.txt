import { z } from 'zod'
import { eq } from 'drizzle-orm'
import { users } from '../../db/schema'

const bodySchema = z
  .object({
    name: z.string().trim().min(1).max(100).optional(),
    email: z.string().trim().email().max(254).optional()
  })
  .refine(v => v.name !== undefined || v.email !== undefined, {
    message: 'At least one of name or email is required'
  })

export default defineEventHandler(async (event) => {
  const session = await requireUserSession(event)

  const id = getRouterParam(event, 'id')
  if (!id || id !== session.user.id) {
    throw createError({ statusCode: 403, statusMessage: 'Forbidden' })
  }

  const body = await readValidatedBody(event, (input) => {
    const result = bodySchema.safeParse(input)
    if (!result.success) {
      throw createError({
        statusCode: 400,
        statusMessage: 'Invalid request body',
        data: result.error.flatten()
      })
    }
    return result.data
  })

  const values: { name?: string, email?: string } = {}
  if (body.name !== undefined) values.name = body.name
  if (body.email !== undefined) values.email = body.email

  const db = useDB()
  const [updated] = await db
    .update(users)
    .set(values)
    .where(eq(users.id, id))
    .returning({
      id: users.id,
      name: users.name,
      email: users.email,
      role: users.role,
      createdAt: users.createdAt
    })

  if (!updated) {
    throw createError({ statusCode: 404, statusMessage: 'User not found' })
  }

  return updated
})

export default defineCachedEventHandler(async (event) => {
  const { user } = await requireUserSession(event)
  const stats = await computeUserStats(user.id)

  return {
    userId: user.id,
    stats,
    generatedAt: new Date().toISOString()
  }
}, {
  name: 'me-dashboard',
  maxAge: 60,
  swr: false,
  getKey: async (event) => {
    const session = await getUserSession(event)
    return session.user?.id ? `user:${session.user.id}` : 'anonymous'
  }
})
