export default defineNuxtPlugin((nuxtApp) => {
  const token = useCookie<string | null>('token')

  const isApiRequest = (request: RequestInfo | URL): boolean => {
    const url = typeof request === 'string'
      ? request
      : request instanceof URL
        ? request.href
        : request.url
    try {
      const parsed = new URL(url, 'http://localhost')
      return parsed.pathname === '/api' || parsed.pathname.startsWith('/api/')
    }
    catch {
      return url.startsWith('/api')
    }
  }

  const baseFetch = import.meta.server ? useRequestFetch() : globalThis.$fetch

  const api = (baseFetch as typeof $fetch).create({
    onRequest({ request, options }) {
      if (!isApiRequest(request)) return
      const value = token.value
      if (!value) return
      const headers = new Headers(options.headers as HeadersInit | undefined)
      headers.set('Authorization', `Bearer ${value}`)
      options.headers = headers
    },
  }) as typeof $fetch

  if (import.meta.client) {
    globalThis.$fetch = api
  }

  nuxtApp.hook('app:created', () => {})

  return {
    provide: {
      api,
    },
  }
})

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui'

interface Order {
  id: string | number
  customer: string
  total: number
  status: string
  createdAt: string
}

interface OrdersResponse {
  items: Order[]
  total: number
}

const PAGE_SIZE = 10

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

const sortOptions = [
  { label: 'Newest', value: '-createdAt' },
  { label: 'Oldest', value: 'createdAt' },
  { label: 'Total (high → low)', value: '-total' },
  { label: 'Total (low → high)', value: 'total' },
  { label: 'Customer', value: 'customer' },
  { label: 'Status', value: 'status' },
]
const defaultSort = sortOptions[0]!.value

const firstQueryValue = (v: unknown): string | undefined => {
  const raw = Array.isArray(v) ? v[0] : v
  return typeof raw === 'string' && raw !== '' ? raw : undefined
}

const page = computed<number>({
  get() {
    const n = Number.parseInt(firstQueryValue(route.query.page) ?? '1', 10)
    return Number.isFinite(n) && n > 0 ? n : 1
  },
  set(value) {
    router.push({ query: { ...route.query, page: String(value) } })
  },
})

const sort = computed<string>({
  get() {
    const s = firstQueryValue(route.query.sort)
    return s && sortOptions.some(o => o.value === s) ? s : defaultSort
  },
  set(value) {
    router.push({ query: { ...route.query, sort: value, page: '1' } })
  },
})

const { data, status, error } = await useFetch<OrdersResponse>('/api/orders', {
  $fetch: $api,
  query: computed(() => ({ page: page.value, sort: sort.value })),
  default: () => ({ items: [], total: 0 }),
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)

const currency = new Intl.NumberFormat(undefined, { style: 'currency', currency: 'USD' })

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'ID' },
  { accessorKey: 'customer', header: 'Customer' },
  { accessorKey: 'status', header: 'Status' },
  {
    accessorKey: 'total',
    header: 'Total',
    cell: ({ row }) => currency.format(Number(row.getValue('total'))),
  },
  {
    accessorKey: 'createdAt',
    header: 'Created',
    cell: ({ row }) => new Date(row.getValue('createdAt') as string).toLocaleString(),
  },
]
</script>

<template>
  <div class="space-y-4 p-4">
    <div class="flex items-center justify-between gap-4">
      <h1 class="text-xl font-semibold">
        Orders
      </h1>
      <USelect
        v-model="sort"
        :items="sortOptions"
        value-key="value"
        class="w-56"
      />
    </div>

    <UAlert
      v-if="error"
      color="error"
      variant="subtle"
      title="Failed to load orders"
      :description="error.message"
    />

    <UTable
      :data="items"
      :columns="columns"
      :loading="status === 'pending'"
    />

    <div class="flex justify-end">
      <UPagination
        v-model:page="page"
        :total="total"
        :items-per-page="PAGE_SIZE"
      />
    </div>
  </div>
</template>
