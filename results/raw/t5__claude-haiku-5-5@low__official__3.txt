import { createHmac, timingSafeEqual } from 'node:crypto'

const SIGNATURE_TOLERANCE_SECONDS = 300

interface StripeCheckoutSession {
  id: string
  payment_status: string
  customer_details?: { email?: string | null } | null
  metadata?: Record<string, string> | null
}

interface StripeEvent {
  id: string
  type: string
  created: number
  data: { object: unknown }
}

function parseSignatureHeader(header: string): { timestamp: string | undefined, signatures: string[] } {
  let timestamp: string | undefined
  const signatures: string[] = []

  for (const element of header.split(',')) {
    const separator = element.indexOf('=')
    if (separator === -1) continue
    const key = element.slice(0, separator).trim()
    const value = element.slice(separator + 1).trim()
    if (key === 't') timestamp = value
    else if (key === 'v1') signatures.push(value)
  }

  return { timestamp, signatures }
}

function verifyStripeSignature(payload: string, header: string, secret: string): boolean {
  const { timestamp, signatures } = parseSignatureHeader(header)
  if (!timestamp || signatures.length === 0) return false

  const timestampSeconds = Number(timestamp)
  if (!Number.isFinite(timestampSeconds)) return false
  if (Math.abs(Math.floor(Date.now() / 1000) - timestampSeconds) > SIGNATURE_TOLERANCE_SECONDS) return false

  const expected = createHmac('sha256', secret)
    .update(`${timestamp}.${payload}`, 'utf8')
    .digest()

  return signatures.some((signature) => {
    if (!/^[0-9a-f]+$/i.test(signature)) return false
    const provided = Buffer.from(signature, 'hex')
    return provided.length === expected.length && timingSafeEqual(provided, expected)
  })
}

async function handleCheckoutSessionCompleted(session: StripeCheckoutSession, eventId: string) {
  if (session.payment_status !== 'paid') {
    console.info(`[stripe] checkout session ${session.id} not paid yet (event ${eventId}), skipping`)
    return
  }

  // Fulfilment must be idempotent: Stripe retries webhooks and may deliver the same event more than once.
  // Key any side effects on session.id (or eventId) before granting access or creating records.
  console.info(`[stripe] checkout session ${session.id} completed (event ${eventId})`)
}

export default defineEventHandler(async (event) => {
  const config = useRuntimeConfig(event)
  const secret = config.stripeWebhookSecret

  if (!secret) {
    throw createError({ statusCode: 500, statusMessage: 'Stripe webhook secret is not configured' })
  }

  const signature = getHeader(event, 'stripe-signature')
  if (!signature) {
    throw createError({ statusCode: 400, statusMessage: 'Missing Stripe-Signature header' })
  }

  // Signature must be verified against the exact raw bytes Stripe sent, so read the body as text before any parsing.
  const payload = await readRawBody(event, 'utf8')
  if (!payload) {
    throw createError({ statusCode: 400, statusMessage: 'Empty request body' })
  }

  if (!verifyStripeSignature(payload, signature, secret)) {
    throw createError({ statusCode: 400, statusMessage: 'Invalid Stripe signature' })
  }

  let stripeEvent: StripeEvent
  try {
    stripeEvent = JSON.parse(payload) as StripeEvent
  }
  catch {
    throw createError({ statusCode: 400, statusMessage: 'Invalid JSON payload' })
  }

  switch (stripeEvent.type) {
    case 'checkout.session.completed':
      await handleCheckoutSessionCompleted(stripeEvent.data.object as StripeCheckoutSession, stripeEvent.id)
      break
    default:
      break
  }

  return { received: true }
})
