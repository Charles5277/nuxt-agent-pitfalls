import type { $Fetch } from 'ofetch'

function isApiRequest(request: RequestInfo | URL): boolean {
  let raw: string
  if (typeof request === 'string') raw = request
  else if (request instanceof URL) raw = request.href
  else raw = request.url

  if (raw.startsWith('/api/') || raw === '/api' || raw.startsWith('/api?')) {
    return true
  }

  try {
    const base = import.meta.client ? window.location.origin : 'http://localhost'
    const url = new URL(raw, base)
    if (import.meta.client && url.origin !== window.location.origin) return false
    return url.pathname === '/api' || url.pathname.startsWith('/api/')
  } catch {
    return false
  }
}

export default defineNuxtPlugin({
  name: 'api',
  enforce: 'pre',
  setup() {
    const token = useCookie<string | null>('token')

    const api = $fetch.create({
      onRequest({ request, options }) {
        if (!isApiRequest(request)) return
        const value = token.value
        if (!value) return
        const headers = new Headers(options.headers as HeadersInit | undefined)
        headers.set('Authorization', `Bearer ${value}`)
        options.headers = headers
      },
    }) as $Fetch

    if (import.meta.client) {
      globalThis.$fetch = api as typeof globalThis.$fetch
    }

    return {
      provide: {
        api,
      },
    }
  },
})

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui'

interface Order {
  id: string | number
  customer: string
  status: string
  total: number
  createdAt: string
}

interface OrdersResponse {
  items: Order[]
  total: number
}

const PAGE_SIZE = 20

const sortOptions = [
  { label: 'Created date', value: 'createdAt' },
  { label: 'Customer', value: 'customer' },
  { label: 'Status', value: 'status' },
  { label: 'Total', value: 'total' },
]
const sortValues = sortOptions.map(o => o.value)
const DEFAULT_SORT = 'createdAt'

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

function firstQueryValue(value: unknown): string | undefined {
  if (Array.isArray(value)) return value[0] ?? undefined
  return typeof value === 'string' ? value : undefined
}

const page = computed<number>({
  get() {
    const n = Number.parseInt(firstQueryValue(route.query.page) ?? '', 10)
    return Number.isFinite(n) && n >= 1 ? n : 1
  },
  set(value) {
    router.push({
      query: {
        ...route.query,
        page: value > 1 ? String(value) : undefined,
      },
    })
  },
})

const sort = computed<string>({
  get() {
    const s = firstQueryValue(route.query.sort)
    return s && sortValues.includes(s) ? s : DEFAULT_SORT
  },
  set(value) {
    router.push({
      query: {
        ...route.query,
        sort: value === DEFAULT_SORT ? undefined : value,
        page: undefined,
      },
    })
  },
})

const { data, status, error, refresh } = await useFetch<OrdersResponse>('/api/orders', {
  $fetch: $api,
  query: { page, sort },
  default: () => ({ items: [], total: 0 }),
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)
const loading = computed(() => status.value === 'pending')

const currency = new Intl.NumberFormat(undefined, { style: 'currency', currency: 'USD' })
const dateFormat = new Intl.DateTimeFormat(undefined, { dateStyle: 'medium', timeStyle: 'short' })

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'Order' },
  { accessorKey: 'customer', header: 'Customer' },
  { accessorKey: 'status', header: 'Status' },
  {
    accessorKey: 'total',
    header: 'Total',
    cell: ({ row }) => currency.format(Number(row.getValue('total'))),
  },
  {
    accessorKey: 'createdAt',
    header: 'Created',
    cell: ({ row }) => {
      const d = new Date(row.getValue('createdAt') as string)
      return Number.isNaN(d.getTime()) ? '' : dateFormat.format(d)
    },
  },
]

useHead({ title: 'Orders' })
</script>

<template>
  <div class="mx-auto flex max-w-6xl flex-col gap-4 p-4">
    <div class="flex flex-wrap items-center justify-between gap-3">
      <h1 class="text-xl font-semibold">
        Orders
      </h1>
      <div class="flex items-center gap-2">
        <span class="text-sm text-muted">Sort by</span>
        <USelect
          v-model="sort"
          :items="sortOptions"
          class="w-44"
        />
      </div>
    </div>

    <UAlert
      v-if="error"
      color="error"
      variant="subtle"
      title="Failed to load orders"
      :description="error.message"
      :actions="[{ label: 'Retry', onClick: () => refresh() }]"
    />

    <UTable
      :data="items"
      :columns="columns"
      :loading="loading"
      class="flex-1"
    />

    <div class="flex items-center justify-between gap-3">
      <span class="text-sm text-muted">{{ total }} orders</span>
      <UPagination
        v-model:page="page"
        :total="total"
        :items-per-page="PAGE_SIZE"
        :disabled="loading"
      />
    </div>
  </div>
</template>
