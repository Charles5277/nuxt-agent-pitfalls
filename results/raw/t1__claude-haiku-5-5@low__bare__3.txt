=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: [user: { id: string; name: string }]
}>()

const typedName = ref('')

const isMatch = computed(() => typedName.value === props.user.name)

watch(
  () => props.open,
  (value) => {
    if (value) typedName.value = ''
  }
)

function close() {
  emit('update:open', false)
}

function onConfirm() {
  if (!isMatch.value) return
  emit('confirm', props.user)
  emit('update:open', false)
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          This action cannot be undone. This will permanently delete the user
          <strong>{{ user.name }}</strong> and remove all associated data.
        </p>
        <div class="space-y-2">
          <label for="delete-user-confirm" class="text-sm">
            Type <strong>{{ user.name }}</strong> to confirm.
          </label>
          <UInput id="delete-user-confirm" v-model="typedName" class="w-full" />
        </div>
      </div>
    </template>
    <template #footer>
      <div class="flex justify-end gap-2 w-full">
        <UButton color="neutral" variant="outline" @click="close">Cancel</UButton>
        <UButton color="error" :disabled="!isMatch" @click="onConfirm">Delete</UButton>
      </div>
    </template>
  </UModal>
</template>
