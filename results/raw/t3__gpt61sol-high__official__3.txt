import { eq } from 'drizzle-orm'
import { createError, defineEventHandler, getRouterParam, readValidatedBody } from 'h3'
import { z } from 'zod'
import { users } from '../../db/schema'

// [Body validation](https://v1.h3.dev/utils/request#readvalidatedbodyevent-validate)
const profileSchema = z.object({
  name: z.string().trim().min(1).max(100).optional(),
  email: z.string().trim().email().max(254).optional()
}).strict().refine(
  body => body.name !== undefined || body.email !== undefined,
  { message: 'Provide at least one profile field' }
)

export default defineEventHandler(async (event) => {
  const { user } = await requireUserSession(event)
  const id = getRouterParam(event, 'id')

  if (!id) {
    throw createError({ statusCode: 400, statusMessage: 'Missing user ID' })
  }

  if (id !== user.id) {
    throw createError({ statusCode: 403, statusMessage: 'Forbidden' })
  }

  const body = await readValidatedBody(event, value => profileSchema.parse(value))

  // [Drizzle updates](https://orm.drizzle.team/docs/update)
  const [updatedUser] = await useDB()
    .update(users)
    .set({
      name: body.name,
      email: body.email
    })
    .where(eq(users.id, user.id))
    .returning({
      id: users.id,
      name: users.name,
      email: users.email
    })

  if (!updatedUser) {
    throw createError({ statusCode: 404, statusMessage: 'User not found' })
  }

  return updatedUser
})

  }
