<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui'

interface Order {
  id: string | number
  createdAt: string
  status: string
  total: number
}

interface OrdersResponse {
  items: Order[]
  total: number
}

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

// Must match the API's fixed page size.
const pageSize = 10

const sortItems = [
  { label: 'Order ID', value: 'id' },
  { label: 'Created at', value: 'createdAt' },
  { label: 'Status', value: 'status' },
  { label: 'Total', value: 'total' }
]

const page = computed<number>({
  get() {
    const raw = route.query.page
    const value = Number(Array.isArray(raw) ? raw[0] : raw)
    return Number.isSafeInteger(value) && value > 0 ? value : 1
  },
  set(value) {
    void router.push({
      query: { ...route.query, page: String(value) },
      hash: route.hash
    })
  }
})

const sort = computed<string>({
  get() {
    const raw = route.query.sort
    const value = Array.isArray(raw) ? raw[0] : raw
    return sortItems.find(item => item.value === value)?.value ?? 'createdAt'
  },
  set(value) {
    void router.push({
      query: { ...route.query, page: '1', sort: value },
      hash: route.hash
    })
  }
})

const query = computed(() => ({
  page: page.value,
  sort: sort.value
}))

const { data, status, error, refresh } = await useFetch<OrdersResponse>(
  '/api/orders',
  {
    $fetch: $api,
    query,
    dedupe: 'cancel',
    default: () => ({ items: [], total: 0 })
  }
)

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'Order ID' },
  { accessorKey: 'createdAt', header: 'Created at' },
  { accessorKey: 'status', header: 'Status' },
  { accessorKey: 'total', header: 'Total' }
]
</script>

<template>
  <main class="mx-auto max-w-6xl space-y-6 p-6">
    <div class="flex flex-wrap items-center justify-between gap-4">
      <h1 class="text-2xl font-semibold">Orders</h1>

      <div class="flex items-center gap-2">
        <label for="orders-sort" class="text-sm">Sort by</label>
        <USelect
          id="orders-sort"
          v-model="sort"
          :items="sortItems"
          class="w-44"
        />
      </div>
    </div>

    <div v-if="error" role="alert" class="flex items-center gap-4">
      <p class="text-error">Unable to load orders.</p>
      <UButton
        label="Retry"
        color="neutral"
        variant="outline"
        :loading="status === 'pending'"
        @click="refresh()"
      />
    </div>

    <UTable
      :data="data.items"
      :columns="columns"
      :loading="status === 'pending'"
      :pagination-options="{ manualPagination: true }"
      :sorting-options="{ manualSorting: true }"
      :empty="error ? 'Orders could not be loaded.' : 'No orders found.'"
      class="w-full"
    />

    <div class="flex flex-wrap items-center justify-between gap-4">
      <p aria-live="polite" class="text-sm text-muted">
        {{ data.total }} orders
      </p>

      <UPagination
        v-model:page="page"
        :total="data.total"
        :items-per-page="pageSize"
      />
    </div>
  </main>
</template>

import { withBase } from 'ufo'

export default defineNuxtPlugin({
  name: 'api',
  enforce: 'pre',
  setup() {
    const token = useCookie<string | null>('token')
    const origin = useRequestURL().origin
    const requestFetch = useRequestFetch()

    // All application API calls must use $api, or useFetch with $fetch: $api.
    // An app plugin cannot intercept arbitrary fetch clients or Nitro requests.
    // https://nuxt.com/docs/4.x/guide/recipes/custom-usefetch
    const api = requestFetch.create({
      onRequest({ request, options }) {
        const target = typeof request === 'string'
          ? withBase(request, options.baseURL || '/')
          : request.url

        const url = new URL(target, origin)
        const isLocalApi = url.origin === origin
          && (url.pathname === '/api' || url.pathname.startsWith('/api/'))

        if (isLocalApi && token.value) {
          const headers = new Headers(options.headers)
          headers.set('Authorization', `Bearer ${token.value}`)
          options.headers = headers
        }
      }
    })

    return {
      provide: { api }
    }
  }
})
