=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: [id: string]
}>()

const confirmation = ref('')

const isMatch = computed(() => confirmation.value === props.user.name)

watch(() => props.open, (isOpen) => {
  if (!isOpen) confirmation.value = ''
})

watch(() => props.user.id, () => {
  confirmation.value = ''
})

function cancel() {
  emit('update:open', false)
}

function confirmDelete() {
  if (!isMatch.value) return
  emit('confirm', props.user.id)
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl sm:max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          Deleting <strong>{{ user.name }}</strong> is permanent and cannot be undone. All data associated with this user will be removed.
        </p>
        <UFormField :label="`Type ${user.name} to confirm`" class="w-full">
          <UInput
            v-model="confirmation"
            autocomplete="off"
            class="w-full"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="cancel">Cancel</UButton>
        <UButton color="error" :disabled="!isMatch" @click="confirmDelete">Delete</UButton>
      </div>
    </template>
  </UModal>
</template>
