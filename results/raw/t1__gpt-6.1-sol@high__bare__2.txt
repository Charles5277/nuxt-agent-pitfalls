=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: {
    id: string
    name: string
  }
}>()

const emit = defineEmits<{
  'update:open': [open: boolean]
  confirm: []
}>()

const typedName = ref('')
const nameMatches = computed(() => typedName.value === props.user.name)

watch(
  () => [props.open, props.user.id, props.user.name] as const,
  () => {
    typedName.value = ''
  },
  { flush: 'sync' }
)

function confirmDeletion() {
  if (!props.open || !nameMatches.value) return

  emit('confirm')
  emit('update:open', false)
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    description="This action permanently deletes the user."
    :ui="{ content: 'max-w-2xl', footer: 'justify-end' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          Deleting <strong>{{ props.user.name }}</strong> cannot be undone.
          Type their exact name below to confirm.
        </p>

        <UInput
          v-model="typedName"
          aria-label="User's exact name"
          autocomplete="off"
          :spellcheck="false"
          class="w-full"
        />
      </div>
    </template>

    <template #footer>
      <UButton
        label="Cancel"
        color="neutral"
        variant="outline"
        @click="emit('update:open', false)"
      />
      <UButton
        label="Delete"
        color="error"
        :disabled="!nameMatches"
        @click="confirmDeletion"
      />
    </template>
  </UModal>
</template>
