=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string, name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const typedName = ref('')

const canDelete = computed(() => typedName.value === props.user.name)

watch(() => props.open, (isOpen) => {
  if (!isOpen) {
    typedName.value = ''
  }
})

function onUpdateOpen(value: boolean) {
  emit('update:open', value)
}

function onCancel() {
  emit('update:open', false)
}

function onConfirm() {
  if (!canDelete.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="onUpdateOpen"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. This will permanently delete the user
          <strong class="font-semibold text-highlighted">{{ user.name }}</strong>
          and remove all of their associated data.
        </p>

        <UFormField
          :label="`Type &quot;${user.name}&quot; to confirm`"
          class="w-full"
        >
          <UInput
            v-model="typedName"
            :placeholder="user.name"
            autocomplete="off"
            class="w-full"
            @keydown.enter="onConfirm"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          label="Cancel"
          color="neutral"
          variant="outline"
          @click="onCancel"
        />
        <UButton
          label="Delete"
          color="error"
          :disabled="!canDelete"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
=== END ===
