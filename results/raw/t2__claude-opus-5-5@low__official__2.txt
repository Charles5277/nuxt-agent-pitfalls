export default defineNuxtPlugin(() => {
  const token = useCookie<string | null>('token')

  const isApiRequest = (request: RequestInfo | URL): boolean => {
    const raw = typeof request === 'string'
      ? request
      : request instanceof URL
        ? request.href
        : request.url
    try {
      const url = new URL(raw, 'http://localhost')
      return url.pathname === '/api' || url.pathname.startsWith('/api/')
    }
    catch {
      return raw.startsWith('/api')
    }
  }

  const api = $fetch.create({
    onRequest({ request, options }) {
      if (!isApiRequest(request) || !token.value) return
      const headers = new Headers(options.headers as HeadersInit | undefined)
      headers.set('Authorization', `Bearer ${token.value}`)
      options.headers = headers
    },
  })

  return {
    provide: {
      api,
    },
  }
})

declare module '#app' {
  interface NuxtApp {
    $api: typeof $fetch
  }
}

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui'

interface Order {
  id: number | string
  customer: string
  total: number
  status: string
  createdAt: string
}

interface OrdersResponse {
  items: Order[]
  total: number
}

const PAGE_SIZE = 20

const sortOptions = [
  { label: 'Newest', value: '-createdAt' },
  { label: 'Oldest', value: 'createdAt' },
  { label: 'Total (high → low)', value: '-total' },
  { label: 'Total (low → high)', value: 'total' },
  { label: 'Customer', value: 'customer' },
  { label: 'Status', value: 'status' },
]
const allowedSorts = sortOptions.map(o => o.value)
const DEFAULT_SORT = '-createdAt'

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

const firstValue = (v: unknown): string | undefined =>
  Array.isArray(v) ? (v[0] ?? undefined) as string | undefined : (v as string | undefined) ?? undefined

const page = computed<number>({
  get() {
    const n = Number.parseInt(firstValue(route.query.page) ?? '1', 10)
    return Number.isFinite(n) && n > 0 ? n : 1
  },
  set(value) {
    router.push({ query: { ...route.query, page: String(value) } })
  },
})

const sort = computed<string>({
  get() {
    const s = firstValue(route.query.sort)
    return s && allowedSorts.includes(s) ? s : DEFAULT_SORT
  },
  set(value) {
    router.push({ query: { ...route.query, sort: value, page: '1' } })
  },
})

const query = computed(() => ({ page: page.value, sort: sort.value }))

const { data, status, error, refresh } = await useFetch<OrdersResponse>('/api/orders', {
  query,
  $fetch: $api,
  default: () => ({ items: [], total: 0 }),
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)

const currency = new Intl.NumberFormat(undefined, { style: 'currency', currency: 'USD' })

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'ID' },
  { accessorKey: 'customer', header: 'Customer' },
  { accessorKey: 'status', header: 'Status' },
  {
    accessorKey: 'total',
    header: 'Total',
    cell: ({ row }) => currency.format(Number(row.getValue('total'))),
  },
  {
    accessorKey: 'createdAt',
    header: 'Created',
    cell: ({ row }) => new Date(row.getValue('createdAt') as string).toLocaleString(),
  },
]
</script>

<template>
  <div class="p-6 space-y-4">
    <div class="flex items-center justify-between gap-4">
      <h1 class="text-xl font-semibold">
        Orders
      </h1>
      <USelect
        v-model="sort"
        :items="sortOptions"
        value-key="value"
        class="w-56"
      />
    </div>

    <UAlert
      v-if="error"
      color="error"
      title="Failed to load orders"
      :description="error.message"
      :actions="[{ label: 'Retry', onClick: () => refresh() }]"
    />

    <UTable
      :data="items"
      :columns="columns"
      :loading="status === 'pending'"
    />

    <div class="flex justify-end">
      <UPagination
        v-model:page="page"
        :total="total"
        :items-per-page="PAGE_SIZE"
      />
    </div>
  </div>
</template>
