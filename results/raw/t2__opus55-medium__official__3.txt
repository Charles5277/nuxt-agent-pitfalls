import type { FetchRequest } from 'ofetch'

/**
 * Provides `$api`: a `$fetch` instance that adds
 * `Authorization: Bearer <token>` to every request aimed at this app's `/api`.
 *
 * Use it directly (`useNuxtApp().$api('/api/...')`) or pass it to useFetch /
 * useAsyncData (`useFetch('/api/...', { $fetch: useNuxtApp().$api })`).
 * The global `$fetch` is not patched. It is shared by every request on the
 * server, so patching it would leak one user's token into another user's request.
 */
export default defineNuxtPlugin(() => {
  const token = useCookie<string | null>('token')
  const origin = useRequestURL().origin

  const isApiPath = (pathname: string) =>
    pathname === '/api' || pathname.startsWith('/api/')

  const isApiRequest = (request: FetchRequest): boolean => {
    const raw = typeof request === 'string' ? request : request.url
    if (raw.startsWith('/')) {
      return isApiPath(raw.split(/[?#]/, 1)[0]!)
    }
    try {
      const url = new URL(raw)
      return url.origin === origin && isApiPath(url.pathname)
    }
    catch {
      return false
    }
  }

  const api = $fetch.create({
    onRequest({ request, options }) {
      if (!token.value || !isApiRequest(request)) return
      const headers = new Headers(options.headers as HeadersInit | undefined)
      headers.set('Authorization', `Bearer ${token.value}`)
      options.headers = headers
    },
  })

  return {
    provide: {
      api,
    },
  }
})

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui'

interface Order {
  id: string | number
  customer: string
  status: string
  total: number
  createdAt: string
}

interface OrdersResponse {
  items: Order[]
  total: number
}

// The API decides the page size. Keep this the same as the server's value.
const PAGE_SIZE = 10

const sortOptions = [
  { label: 'Newest first', value: '-createdAt' },
  { label: 'Oldest first', value: 'createdAt' },
  { label: 'Total: high to low', value: '-total' },
  { label: 'Total: low to high', value: 'total' },
  { label: 'Customer (A–Z)', value: 'customer' },
  { label: 'Status', value: 'status' },
]
const allowedSorts = sortOptions.map(o => o.value)
const DEFAULT_SORT = sortOptions[0]!.value

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

const firstQueryValue = (v: unknown): string | undefined => {
  if (Array.isArray(v)) return v[0] == null ? undefined : String(v[0])
  return v == null ? undefined : String(v)
}

const page = computed<number>({
  get() {
    const n = Number.parseInt(firstQueryValue(route.query.page) ?? '1', 10)
    return Number.isFinite(n) && n > 0 ? n : 1
  },
  set(value) {
    router.push({
      query: { ...route.query, page: value > 1 ? String(value) : undefined },
    })
  },
})

const sort = computed<string>({
  get() {
    const s = firstQueryValue(route.query.sort)
    return s && allowedSorts.includes(s) ? s : DEFAULT_SORT
  },
  set(value) {
    router.push({
      query: {
        ...route.query,
        sort: value === DEFAULT_SORT ? undefined : value,
        page: undefined,
      },
    })
  },
})

const { data, status, error, refresh } = await useFetch<OrdersResponse>('/api/orders', {
  $fetch: $api,
  query: computed(() => ({ page: page.value, sort: sort.value })),
  default: () => ({ items: [], total: 0 }),
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)

const currency = new Intl.NumberFormat(undefined, { style: 'currency', currency: 'USD' })
const dateFormat = new Intl.DateTimeFormat(undefined, { dateStyle: 'medium', timeStyle: 'short' })

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'Order' },
  { accessorKey: 'customer', header: 'Customer' },
  { accessorKey: 'status', header: 'Status' },
  {
    accessorKey: 'total',
    header: () => h('div', { class: 'text-right' }, 'Total'),
    cell: ({ row }) => h('div', { class: 'text-right font-medium' }, currency.format(Number(row.getValue('total')))),
  },
  {
    accessorKey: 'createdAt',
    header: 'Created',
    cell: ({ row }) => {
      const d = new Date(String(row.getValue('createdAt')))
      return Number.isNaN(d.getTime()) ? String(row.getValue('createdAt')) : dateFormat.format(d)
    },
  },
]

useSeoMeta({ title: 'Orders' })
</script>

<template>
  <UContainer class="py-8 space-y-4">
    <div class="flex flex-wrap items-center justify-between gap-4">
      <h1 class="text-2xl font-semibold">
        Orders
      </h1>

      <USelect
        v-model="sort"
        :items="sortOptions"
        value-key="value"
        icon="i-lucide-arrow-up-down"
        class="w-56"
        aria-label="Sort orders"
      />
    </div>

    <UAlert
      v-if="error"
      color="error"
      variant="subtle"
      icon="i-lucide-circle-alert"
      title="Failed to load orders"
      :description="error.message"
      :actions="[{ label: 'Retry', onClick: () => refresh() }]"
    />

    <UTable
      :data="items"
      :columns="columns"
      :loading="status === 'pending'"
      class="border border-default rounded-lg"
    >
      <template #empty>
        <div class="py-6 text-center text-muted">
          No orders found.
        </div>
      </template>
    </UTable>

    <div class="flex items-center justify-between gap-4">
      <p class="text-sm text-muted">
        {{ total }} order{{ total === 1 ? '' : 's' }}
      </p>

      <UPagination
        v-if="total > PAGE_SIZE"
        v-model:page="page"
        :total="total"
        :items-per-page="PAGE_SIZE"
      />
    </div>
  </UContainer>
</template>
