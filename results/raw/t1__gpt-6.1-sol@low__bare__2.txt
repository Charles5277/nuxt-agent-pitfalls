=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: {
    id: string
    name: string
  }
}>()

const emit = defineEmits<{
  'update:open': [open: boolean]
  confirm: []
}>()

const typedName = ref('')
const nameMatches = computed(() => typedName.value === props.user.name)

watch(
  () => [props.open, props.user.id, props.user.name] as const,
  () => {
    typedName.value = ''
  }
)

function close() {
  emit('update:open', false)
}

function confirm() {
  if (nameMatches.value) {
    emit('confirm')
  }
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          Deleting {{ props.user.name }} is permanent and cannot be undone.
          Type their exact name to confirm.
        </p>

        <UFormField label="User name">
          <UInput
            v-model="typedName"
            class="w-full"
            autocomplete="off"
            :spellcheck="false"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="close">
          Cancel
        </UButton>
        <UButton color="error" :disabled="!nameMatches" @click="confirm">
          Delete
        </UButton>
      </div>
    </template>
  </UModal>
</template>
=== END ===
