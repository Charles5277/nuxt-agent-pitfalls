=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const confirmation = ref('')

const matches = computed(() => confirmation.value === props.user.name)

watch(
  () => props.open,
  (isOpen) => {
    if (!isOpen) {
      confirmation.value = ''
    }
  }
)

function onUpdateOpen(value: boolean) {
  emit('update:open', value)
}

function onCancel() {
  emit('update:open', false)
}

function onConfirm() {
  if (!matches.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    :content="{ 'aria-describedby': undefined }"
    @update:open="onUpdateOpen"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          You are about to permanently delete
          <strong class="text-highlighted">{{ user.name }}</strong>.
          This action cannot be undone. To confirm, type the user's name
          exactly as shown above.
        </p>

        <UInput
          v-model="confirmation"
          class="w-full"
          :placeholder="user.name"
          autocomplete="off"
          aria-label="Type the user's name to confirm deletion"
          @keydown.enter="onConfirm"
        />
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          label="Cancel"
          color="neutral"
          variant="outline"
          @click="onCancel"
        />
        <UButton
          label="Delete"
          color="error"
          :disabled="!matches"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
