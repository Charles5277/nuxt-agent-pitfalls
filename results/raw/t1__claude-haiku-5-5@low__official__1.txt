=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string, name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: []
}>()

const typed = ref('')
const matches = computed(() => typed.value === props.user.name)

watch(() => props.open, (value) => {
  if (!value) typed.value = ''
})

function close() {
  emit('update:open', false)
}

function onConfirm() {
  if (!matches.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          This will permanently delete <strong>{{ user.name }}</strong> and all of its associated data. This action cannot be undone.
        </p>
        <div class="space-y-2">
          <label for="delete-user-confirm" class="text-sm">
            Type <strong>{{ user.name }}</strong> to confirm.
          </label>
          <UInput
            id="delete-user-confirm"
            v-model="typed"
            :placeholder="user.name"
            class="w-full"
          />
        </div>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="close">
          Cancel
        </UButton>
        <UButton color="error" :disabled="!matches" @click="onConfirm">
          Delete
        </UButton>
      </div>
    </template>
  </UModal>
</template>
