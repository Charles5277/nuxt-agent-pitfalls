import { eq } from 'drizzle-orm'
import { z } from 'zod'
import { users } from '../../db/schema'

const profileSchema = z
  .object({
    name: z.string().trim().min(1).max(100).optional(),
    email: z.string().trim().email().max(254).optional(),
  })
  .strict()
  .refine(
    body => body.name !== undefined || body.email !== undefined,
    { message: 'Provide at least one of name or email' },
  )

export default defineEventHandler(async (event) => {
  const { user } = await requireUserSession(event)
  const id = getRouterParam(event, 'id')

  if (!id) {
    throw createError({
      statusCode: 400,
      statusMessage: 'Missing user ID',
    })
  }

  if (id !== user.id) {
    throw createError({
      statusCode: 403,
      statusMessage: 'Forbidden',
    })
  }

  // [H3 body validation](https://v1.h3.dev/utils/request#readvalidatedbodyevent-validate)
  const body = await readValidatedBody(event, data => profileSchema.parse(data))

  const changes: { name?: string; email?: string } = {}

  if (body.name !== undefined) {
    changes.name = body.name
  }

  if (body.email !== undefined) {
    changes.email = body.email
  }

  const [updatedUser] = await useDB()
    .update(users)
    .set(changes)
    .where(eq(users.id, user.id))
    .returning({
      id: users.id,
      name: users.name,
      email: users.email,
    })

  if (!updatedUser) {
    throw createError({
      statusCode: 404,
      statusMessage: 'User not found',
    })
  }

  setResponseHeader(event, 'Cache-Control', 'private, no-store')

  return updatedUser
})

import { createHash } from 'node:crypto'
import type { H3Event } from 'h3'

// [Nitro function caching](https://v2.nitro.build/guide/cache#cached-functions)
const cachedUserStats = defineCachedFunction(
  async (_event: H3Event, userId: string) => computeUserStats(userId),
  {
    name: 'me-dashboard-stats',
    maxAge: 60,
    swr: false,
    getKey: (_event: H3Event, userId: string) =>
      createHash('sha256').update(userId).digest('hex'),
  },
)

export default defineEventHandler(async (event) => {
  const { user } = await requireUserSession(event)

  setResponseHeader(event, 'Cache-Control', 'private, no-store')

  return cachedUserStats(event, user.id)
})
