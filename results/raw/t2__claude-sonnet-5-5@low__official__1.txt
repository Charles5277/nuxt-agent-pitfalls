<script setup lang="ts">
interface Order {
  id: number | string
  [key: string]: any
}

const route = useRoute()
const router = useRouter()

const sortItems = [
  { label: 'Newest', value: 'createdAt' },
  { label: 'Total', value: 'total' },
  { label: 'Status', value: 'status' },
  { label: 'ID', value: 'id' }
]

const page = computed({
  get: () => Math.max(1, Number(route.query.page) || 1),
  set: (value: number) => {
    router.push({ query: { ...route.query, page: String(value) } })
  }
})

const sort = computed({
  get: () => (route.query.sort as string) || 'createdAt',
  set: (value: string) => {
    router.push({ query: { ...route.query, sort: value, page: '1' } })
  }
})

const { data, status } = await useFetch<{ items: Order[], total: number }>('/api/orders', {
  query: { page, sort },
  watch: [page, sort]
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)

const columns = [
  { accessorKey: 'id', header: 'ID' },
  { accessorKey: 'customer', header: 'Customer' },
  { accessorKey: 'status', header: 'Status' },
  { accessorKey: 'total', header: 'Total' },
  { accessorKey: 'createdAt', header: 'Created' }
]
</script>

<template>
  <UContainer class="py-6 space-y-4">
    <div class="flex items-center justify-between gap-4">
      <h1 class="text-xl font-semibold">
        Orders
      </h1>
      <USelect v-model="sort" :items="sortItems" class="w-48" />
    </div>

    <UTable :data="items" :columns="columns" :loading="status === 'pending'" />

    <div class="flex justify-center">
      <UPagination v-model:page="page" :total="total" :items-per-page="10" />
    </div>
  </UContainer>
</template>

export default defineNuxtPlugin(() => {
  const token = useCookie('token')

  const api = $fetch.create({
    onRequest({ request, options }) {
      const url = typeof request === 'string' ? request : (request as Request).url
      if (!url.startsWith('/api')) return
      if (token.value) {
        const headers = new Headers(options.headers as HeadersInit | undefined)
        headers.set('Authorization', `Bearer ${token.value}`)
        options.headers = headers
      }
    }
  })

  globalThis.$fetch = api as typeof $fetch

  return {
    provide: { api }
  }
})
