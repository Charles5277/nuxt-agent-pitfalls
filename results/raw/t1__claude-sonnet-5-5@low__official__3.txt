=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const typedName = ref('')

const matches = computed(() => typedName.value === props.user.name)

watch(() => props.open, (isOpen) => {
  if (!isOpen) typedName.value = ''
})

function onConfirm() {
  if (!matches.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. This will permanently delete the user
          <strong class="text-highlighted">{{ user.name }}</strong>
          and all associated data.
        </p>

        <UFormField :label="`Type &quot;${user.name}&quot; to confirm`">
          <UInput
            v-model="typedName"
            class="w-full"
            :placeholder="user.name"
            autocomplete="off"
            @keydown.enter="onConfirm"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          color="neutral"
          variant="outline"
          label="Cancel"
          @click="emit('update:open', false)"
        />
        <UButton
          color="error"
          label="Delete"
          :disabled="!matches"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
=== END ===
