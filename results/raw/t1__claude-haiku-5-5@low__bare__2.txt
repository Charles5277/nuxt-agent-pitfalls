=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: [user: { id: string; name: string }]
}>()

const typedName = ref('')

const matches = computed(() => typedName.value === props.user.name)

watch(
  () => props.open,
  (isOpen) => {
    if (!isOpen) typedName.value = ''
  }
)

function close() {
  emit('update:open', false)
}

function onConfirm() {
  if (!matches.value) return
  emit('confirm', props.user)
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          You are about to permanently delete <strong>{{ user.name }}</strong>.
          This action cannot be undone and all data associated with this user will be removed.
        </p>
        <p>
          To confirm, type <strong>{{ user.name }}</strong> below.
        </p>
        <UInput
          v-model="typedName"
          :placeholder="user.name"
          class="w-full"
          autocomplete="off"
        />
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="close">
          Cancel
        </UButton>
        <UButton
          color="error"
          :disabled="!matches"
          @click="onConfirm"
        >
          Delete
        </UButton>
      </div>
    </template>
  </UModal>
</template>
