=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: [user: { id: string; name: string }]
}>()

const confirmName = ref('')

const isMatch = computed(() => confirmName.value === props.user.name)

watch(
  () => props.open,
  (value) => {
    if (value) confirmName.value = ''
  }
)

watch(
  () => props.user.id,
  () => {
    confirmName.value = ''
  }
)

function setOpen(value: boolean) {
  emit('update:open', value)
}

function onCancel() {
  setOpen(false)
}

function onConfirm() {
  if (!isMatch.value) return
  emit('confirm', props.user)
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    :ui="{ content: 'sm:max-w-2xl' }"
    @update:open="setOpen"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. This will permanently delete the user
          <strong class="font-semibold text-highlighted">{{ props.user.name }}</strong>
          and all associated data. To confirm, please type the user's name below.
        </p>
        <UInput
          v-model="confirmName"
          :placeholder="props.user.name"
          autocomplete="off"
          class="w-full"
          @keydown.enter.prevent="onConfirm"
        />
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" label="Cancel" @click="onCancel" />
        <UButton color="error" label="Delete" :disabled="!isMatch" @click="onConfirm" />
      </div>
    </template>
  </UModal>
</template>
