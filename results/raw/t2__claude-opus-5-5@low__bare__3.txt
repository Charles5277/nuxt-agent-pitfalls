export default defineNuxtPlugin((nuxtApp) => {
  const token = useCookie<string | null>('token')

  const isApiRequest = (request: RequestInfo | URL): boolean => {
    const url
      = typeof request === 'string'
        ? request
        : request instanceof URL
          ? request.href
          : request.url

    if (url.startsWith('/api/') || url === '/api') return true

    try {
      const base = import.meta.client ? window.location.origin : 'http://localhost'
      const parsed = new URL(url, base)
      if (import.meta.client && parsed.origin !== window.location.origin) return false
      return parsed.pathname === '/api' || parsed.pathname.startsWith('/api/')
    }
    catch {
      return false
    }
  }

  const api = $fetch.create({
    onRequest({ request, options }) {
      if (!isApiRequest(request)) return
      const value = token.value
      if (!value) return
      const headers = new Headers(options.headers as HeadersInit | undefined)
      headers.set('Authorization', `Bearer ${value}`)
      options.headers = headers
    },
    onResponseError({ response }) {
      if (response.status === 401) {
        token.value = null
      }
    },
  })

  // Make plain `$fetch` / `useFetch` / `useAsyncData` calls in the browser go
  // through the authenticated fetcher as well. On the server, globalThis is
  // shared across requests, so there only `$api` (bound per request) is safe.
  if (import.meta.client) {
    globalThis.$fetch = api as typeof globalThis.$fetch
  }

  nuxtApp.provide('api', api)
})

  interface NuxtApp {
    $api: typeof $fetch
  }
}

  interface ComponentCustomProperties {
    $api: typeof $fetch
  }

<script setup lang="ts">
import type { TableColumn, SelectItem } from '@nuxt/ui'

interface Order {
  id: string | number
  customer: string
  status: string
  total: number
  createdAt: string
}

interface OrdersResponse {
  items: Order[]
  total: number
}

const PAGE_SIZE = 10

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

const sortOptions: SelectItem[] = [
  { label: 'Newest first', value: '-createdAt' },
  { label: 'Oldest first', value: 'createdAt' },
  { label: 'Total (high → low)', value: '-total' },
  { label: 'Total (low → high)', value: 'total' },
  { label: 'Customer (A → Z)', value: 'customer' },
]
const allowedSorts = sortOptions.map(o => (o as { value: string }).value)
const DEFAULT_SORT = '-createdAt'

function firstQueryValue(v: unknown): string | undefined {
  if (Array.isArray(v)) return v[0] ?? undefined
  return typeof v === 'string' ? v : undefined
}

const page = computed<number>({
  get() {
    const n = Number.parseInt(firstQueryValue(route.query.page) ?? '1', 10)
    return Number.isFinite(n) && n > 0 ? n : 1
  },
  set(value) {
    router.push({ query: { ...route.query, page: value > 1 ? String(value) : undefined } })
  },
})

const sort = computed<string>({
  get() {
    const s = firstQueryValue(route.query.sort)
    return s && allowedSorts.includes(s) ? s : DEFAULT_SORT
  },
  set(value) {
    router.push({
      query: {
        ...route.query,
        sort: value === DEFAULT_SORT ? undefined : value,
        page: undefined,
      },
    })
  },
})

const { data, status, error, refresh } = await useFetch<OrdersResponse>('/api/orders', {
  key: 'orders-list',
  $fetch: $api,
  query: computed(() => ({ page: page.value, sort: sort.value })),
  default: () => ({ items: [], total: 0 }),
  watch: [page, sort],
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)
const loading = computed(() => status.value === 'pending')

const currency = new Intl.NumberFormat(undefined, { style: 'currency', currency: 'USD' })
const dateFmt = new Intl.DateTimeFormat(undefined, { dateStyle: 'medium', timeStyle: 'short' })

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'Order' },
  { accessorKey: 'customer', header: 'Customer' },
  { accessorKey: 'status', header: 'Status' },
  {
    accessorKey: 'total',
    header: 'Total',
    cell: ({ row }) => currency.format(Number(row.getValue('total'))),
  },
  {
    accessorKey: 'createdAt',
    header: 'Created',
    cell: ({ row }) => {
      const d = new Date(row.getValue('createdAt') as string)
      return Number.isNaN(d.getTime()) ? '—' : dateFmt.format(d)
    },
  },
]
</script>

<template>
  <UContainer class="py-8 space-y-4">
    <div class="flex items-center justify-between gap-4">
      <h1 class="text-2xl font-semibold">
        Orders
      </h1>
      <USelect
        v-model="sort"
        :items="sortOptions"
        class="w-56"
        aria-label="Sort orders"
      />
    </div>

    <UAlert
      v-if="error"
      color="error"
      variant="subtle"
      title="Failed to load orders"
      :description="error.message"
      :actions="[{ label: 'Retry', onClick: () => refresh() }]"
    />

    <UTable
      :data="items"
      :columns="columns"
      :loading="loading"
      empty="No orders found."
    />

    <div class="flex items-center justify-between">
      <span class="text-sm text-muted">{{ total }} orders</span>
      <UPagination
        v-model:page="page"
        :total="total"
        :items-per-page="PAGE_SIZE"
      />
    </div>
  </UContainer>
</template>
