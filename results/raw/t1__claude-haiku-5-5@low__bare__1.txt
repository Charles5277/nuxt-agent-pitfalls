=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: []
}>()

const typedName = ref('')

const isMatch = computed(
  () => props.user.name.length > 0 && typedName.value === props.user.name
)

watch(
  () => props.open,
  (value) => {
    if (!value) typedName.value = ''
  }
)

function close() {
  emit('update:open', false)
}

function onConfirm() {
  if (!isMatch.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          This action is permanent and cannot be undone. Deleting <strong>{{ user.name }}</strong> will
          remove their account and all associated data.
        </p>
        <p>
          To confirm, type <strong>{{ user.name }}</strong> below.
        </p>
        <UInput
          v-model="typedName"
          :placeholder="user.name"
          class="w-full"
          autocomplete="off"
        />
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="close">
          Cancel
        </UButton>
        <UButton color="error" :disabled="!isMatch" @click="onConfirm">
          Delete
        </UButton>
      </div>
    </template>
  </UModal>
</template>
