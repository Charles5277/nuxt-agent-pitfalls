=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const confirmation = ref('')

const matches = computed(() => confirmation.value === props.user.name)

const isOpen = computed({
  get: () => props.open,
  set: (value: boolean) => emit('update:open', value)
})

watch(() => props.open, (value) => {
  if (!value) confirmation.value = ''
})

function onCancel() {
  emit('update:open', false)
}

function onConfirm() {
  if (!matches.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    v-model:open="isOpen"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. This will permanently delete the user
          <strong class="text-highlighted">{{ user.name }}</strong>
          and all associated data. To confirm, type
          <strong class="text-highlighted">{{ user.name }}</strong>
          below.
        </p>
        <UInput
          v-model="confirmation"
          class="w-full"
          :placeholder="user.name"
          autocomplete="off"
          @keydown.enter="onConfirm"
        />
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          label="Cancel"
          color="neutral"
          variant="outline"
          @click="onCancel"
        />
        <UButton
          label="Delete"
          color="error"
          :disabled="!matches"
          @click="onConfirm"
        />
      </div>
    </template>
  </UModal>
</template>
=== END ===
