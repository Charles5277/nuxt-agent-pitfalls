import { z } from 'zod'
import { eq, and, ne } from 'drizzle-orm'
import { users } from '~~/server/db/schema'

const bodySchema = z.object({
  name: z.string().trim().min(1).max(100).optional(),
  email: z.string().trim().toLowerCase().email().max(255).optional(),
}).strict().refine(d => d.name !== undefined || d.email !== undefined, {
  message: 'At least one field (name, email) is required',
})

export default defineEventHandler(async (event) => {
  const { user } = await requireUserSession(event)
  const id = getRouterParam(event, 'id')

  if (!id) {
    throw createError({ statusCode: 400, statusMessage: 'Missing user id' })
  }
  if (id !== user.id) {
    throw createError({ statusCode: 403, statusMessage: 'You can only update your own profile' })
  }

  const body = await readValidatedBody(event, bodySchema.parse)
  const db = useDB()

  if (body.email) {
    const existing = await db
      .select({ id: users.id })
      .from(users)
      .where(and(eq(users.email, body.email), ne(users.id, id)))
      .get()
    if (existing) {
      throw createError({ statusCode: 409, statusMessage: 'Email already in use' })
    }
  }

  const updated = await db
    .update(users)
    .set({
      ...(body.name !== undefined && { name: body.name }),
      ...(body.email !== undefined && { email: body.email }),
    })
    .where(eq(users.id, id))
    .returning({ id: users.id, name: users.name, email: users.email, role: users.role })
    .get()

  if (!updated) {
    throw createError({ statusCode: 404, statusMessage: 'User not found' })
  }

  return updated
})
