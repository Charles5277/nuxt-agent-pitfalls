=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [open: boolean]
  confirm: []
}>()

const typedName = ref('')
const nameMatches = computed(() => typedName.value === props.user.name)

watch(
  () => [props.open, props.user.id, props.user.name] as const,
  () => {
    typedName.value = ''
  }
)

function confirmDelete() {
  if (nameMatches.value) {
    emit('confirm')
  }
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    description="Confirm deletion by typing the user's exact name."
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p>
          Warning: Deleting {{ props.user.name }} is permanent and cannot be
          undone. Type their exact name below to confirm.
        </p>

        <UFormField label="User name">
          <UInput
            v-model="typedName"
            class="w-full"
            autocomplete="off"
            :spellcheck="false"
            @keydown.enter.prevent="confirmDelete"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          label="Cancel"
          color="neutral"
          variant="outline"
          @click="emit('update:open', false)"
        />
        <UButton
          label="Delete"
          color="error"
          :disabled="!nameMatches"
          @click="confirmDelete"
        />
      </div>
    </template>
  </UModal>
</template>
=== END ===
