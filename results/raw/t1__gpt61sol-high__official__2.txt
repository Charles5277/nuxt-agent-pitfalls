=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [open: boolean]
  confirm: [userId: string]
}>()

const typedName = ref('')
const nameMatches = computed(() => typedName.value === props.user.name)

watch(
  [() => props.open, () => props.user.id, () => props.user.name],
  () => {
    typedName.value = ''
  },
  { flush: 'sync' }
)

function confirmDelete() {
  if (!props.open || !nameMatches.value) return

  emit('confirm', props.user.id)
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    :description="`Deleting ${props.user.name} is permanent and cannot be undone.`"
    :ui="{ content: 'max-w-2xl', footer: 'justify-end' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <UFormField :label="`Type ${props.user.name} exactly to confirm`">
        <UInput
          v-model="typedName"
          autocomplete="off"
          autocapitalize="off"
          :spellcheck="false"
          class="w-full"
        />
      </UFormField>
    </template>

    <template #footer>
      <UButton
        label="Cancel"
        color="neutral"
        variant="outline"
        @click="emit('update:open', false)"
      />
      <UButton
        label="Delete"
        color="error"
        :disabled="!nameMatches"
        @click="confirmDelete"
      />
    </template>
  </UModal>
</template>
