=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [open: boolean]
  confirm: []
}>()

const typedName = ref('')
const nameMatches = computed(() => typedName.value === props.user.name)

watch(
  [() => props.open, () => props.user.id, () => props.user.name],
  () => {
    typedName.value = ''
  }
)

function confirmDeletion() {
  if (props.open && nameMatches.value) {
    emit('confirm')
  }
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl', footer: 'justify-end' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          Deleting <strong class="text-highlighted">{{ props.user.name }}</strong>
          is permanent and cannot be undone. Type their exact name below to
          confirm.
        </p>

        <UInput
          v-model="typedName"
          aria-label="Type the user's exact name to confirm deletion"
          autocomplete="off"
          :spellcheck="false"
          class="w-full"
        />
      </div>
    </template>

    <template #footer>
      <UButton
        type="button"
        color="neutral"
        variant="outline"
        @click="emit('update:open', false)"
      >
        Cancel
      </UButton>

      <UButton
        type="button"
        color="error"
        :disabled="!nameMatches"
        @click="confirmDeletion"
      >
        Delete
      </UButton>
    </template>
  </UModal>
</template>
=== END ===
