import type { FetchContext } from 'ofetch'

function resolvePath(request: FetchContext['request']): string {
  const raw = typeof request === 'string'
    ? request
    : request instanceof URL
      ? request.toString()
      : request.url

  if (raw.startsWith('/')) {
    return raw
  }

  try {
    const url = new URL(raw)
    if (import.meta.client && url.origin !== window.location.origin) {
      return ''
    }
    return url.pathname
  }
  catch {
    return ''
  }
}

function isApiRequest(request: FetchContext['request']): boolean {
  const path = resolvePath(request)
  return path === '/api' || path.startsWith('/api/') || path.startsWith('/api?')
}

export default defineNuxtPlugin({
  name: 'api',
  enforce: 'pre',
  setup() {
    const token = useCookie<string | null>('token')

    const onRequest = ({ request, options }: FetchContext) => {
      if (!isApiRequest(request)) {
        return
      }
      const value = token.value
      if (!value) {
        return
      }
      const headers = new Headers(options.headers as HeadersInit | undefined)
      if (!headers.has('Authorization')) {
        headers.set('Authorization', `Bearer ${value}`)
      }
      options.headers = headers
    }

    const baseFetch = globalThis.$fetch

    const api = baseFetch.create({
      onRequest,
    }) as typeof $fetch

    if (import.meta.client) {
      globalThis.$fetch = api
    }

    return {
      provide: {
        api,
      },
    }
  },
})

import type { TableColumn, SelectItem } from '@nuxt/ui'

interface Order {
  id: string | number
  customer: string
  status: string
  total: number
  createdAt: string
}

interface OrdersResponse {
  items: Order[]
  total: number
}

const PAGE_SIZE = 20

const SORT_FIELDS = ['createdAt', 'total', 'status', 'customer'] as const
type SortField = (typeof SORT_FIELDS)[number]
const DEFAULT_SORT: SortField = 'createdAt'

const sortItems: SelectItem[] = [
  { label: 'Created date', value: 'createdAt' },
  { label: 'Total', value: 'total' },
  { label: 'Status', value: 'status' },
  { label: 'Customer', value: 'customer' },
]

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

function firstQueryValue(value: unknown): string | undefined {
  if (Array.isArray(value)) {
    return typeof value[0] === 'string' ? value[0] : undefined
  }
  return typeof value === 'string' ? value : undefined
}

function updateQuery(patch: { page?: number, sort?: SortField }) {
  const query = { ...route.query }
  if (patch.page !== undefined) {
    query.page = patch.page > 1 ? String(patch.page) : undefined
  }
  if (patch.sort !== undefined) {
    query.sort = patch.sort !== DEFAULT_SORT ? patch.sort : undefined
  }
  for (const key of Object.keys(query)) {
    if (query[key] === undefined) {
      delete query[key]
    }
  }
  router.push({ query })
}

const page = computed<number>({
  get() {
    const parsed = Number.parseInt(firstQueryValue(route.query.page) ?? '', 10)
    return Number.isFinite(parsed) && parsed >= 1 ? parsed : 1
  },
  set(value) {
    updateQuery({ page: value })
  },
})

const sort = computed<SortField>({
  get() {
    const value = firstQueryValue(route.query.sort)
    return (SORT_FIELDS as readonly string[]).includes(value ?? '')
      ? (value as SortField)
      : DEFAULT_SORT
  },
  set(value) {
    if (value === sort.value) {
      return
    }
    updateQuery({ sort: value, page: 1 })
  },
})

const { data, status, error, refresh } = await useFetch<OrdersResponse>('/api/orders', {
  $fetch: $api,
  query: computed(() => ({
    page: page.value,
    sort: sort.value,
  })),
  default: () => ({ items: [], total: 0 }),
})

const items = computed(() => data.value?.items ?? [])
const total = computed(() => data.value?.total ?? 0)
const loading = computed(() => status.value === 'pending')

watch(total, (value) => {
  const lastPage = Math.max(1, Math.ceil(value / PAGE_SIZE))
  if (value > 0 && page.value > lastPage) {

})
