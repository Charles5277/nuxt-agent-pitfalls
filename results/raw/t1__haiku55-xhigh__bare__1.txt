=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string, name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  confirm: []
}>()

const confirmation = ref('')

const isOpen = computed({
  get: () => props.open,
  set: (value: boolean) => emit('update:open', value)
})

const isMatch = computed(() => confirmation.value === props.user.name)

const confirmLabel = computed(() => `Type ${props.user.name} to confirm`)

watch(() => props.open, (open) => {
  if (!open) confirmation.value = ''
})

watch(() => props.user.id, () => {
  confirmation.value = ''
})

function onCancel() {
  emit('update:open', false)
}

function onConfirm() {
  if (!isMatch.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    v-model:open="isOpen"
    title="Delete user"
    :ui="{ content: 'max-w-2xl sm:max-w-2xl' }"
  >
    <template #body>
      <div class="space-y-6">
        <p class="text-sm text-muted">
          Warning: deleting <strong class="text-highlighted">{{ user.name }}</strong> will permanently remove this user and all of their data. This action cannot be undone.
        </p>
        <UFormField :label="confirmLabel" class="w-full">
          <UInput v-model="confirmation" autocomplete="off" class="w-full" />
        </UFormField>
      </div>
    </template>
    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton color="neutral" variant="outline" @click="onCancel">Cancel</UButton>
        <UButton color="error" :disabled="!isMatch" @click="onConfirm">Delete</UButton>
      </div>
    </template>
  </UModal>
</template>
=== END ===
