=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
const props = defineProps<{
  open: boolean
  user: { id: string, name: string }
}>()

const emit = defineEmits<{
  'update:open': [value: boolean]
  'confirm': []
}>()

const typedName = ref('')
const isMatch = computed(() => typedName.value === props.user.name)

watch(() => props.open, (isOpen) => {
  if (!isOpen) typedName.value = ''
})

function cancel() {
  emit('update:open', false)
}

function confirmDelete() {
  if (!isMatch.value) return
  emit('confirm')
}
</script>

<template>
  <UModal
    :open="open"
    title="Delete user"
    :ui="{ content: 'max-w-2xl' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          This action cannot be undone. The user <strong>{{ user.name }}</strong> and all associated data will be permanently deleted.
        </p>

        <UFormField :label="`Type ${user.name} to confirm`">
          <UInput
            v-model="typedName"
            :placeholder="user.name"
            class="w-full"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <div class="flex w-full justify-end gap-2">
        <UButton
          color="neutral"
          variant="outline"
          @click="cancel"
        >
          Cancel
        </UButton>
        <UButton
          color="error"
          :disabled="!isMatch"
          @click="confirmDelete"
        >
          Delete
        </UButton>
      </div>
    </template>
  </UModal>
</template>
=== END ===
