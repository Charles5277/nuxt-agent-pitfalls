import { z } from 'zod'
import { and, eq, ne } from 'drizzle-orm'
import { users } from '~~/server/db/schema'

const paramsSchema = z.object({
  id: z.string().min(1)
})

const bodySchema = z
  .object({
    name: z.string().trim().min(1).max(100).optional(),
    email: z.string().trim().toLowerCase().email().max(255).optional()
  })
  .strict()
  .refine(data => data.name !== undefined || data.email !== undefined, {
    message: 'At least one of name or email must be provided'
  })

export default defineEventHandler(async (event) => {
  const { user } = await requireUserSession(event)
  const { id } = await getValidatedRouterParams(event, paramsSchema.parse)

  if (user.id !== id) {
    throw createError({ statusCode: 403, statusMessage: 'You can only update your own profile' })
  }

  const body = await readValidatedBody(event, bodySchema.parse)
  const db = useDB()

  if (body.email !== undefined) {
    const conflict = await db
      .select({ id: users.id })
      .from(users)
      .where(and(eq(users.email, body.email), ne(users.id, id)))
      .get()

    if (conflict) {
      throw createError({ statusCode: 409, statusMessage: 'Email is already in use' })
    }
  }

  const updates: Partial<Pick<typeof users.$inferInsert, 'name' | 'email'>> = {}
  if (body.name !== undefined) updates.name = body.name
  if (body.email !== undefined) updates.email = body.email

  const updated = await db
    .update(users)
    .set(updates)
    .where(eq(users.id, id))
    .returning({
      id: users.id,
      name: users.name,
      email: users.email,
      role: users.role,
      createdAt: users.createdAt
    })
    .get()

  if (!updated) {
    throw createError({ statusCode: 404, statusMessage: 'User not found' })
  }

  return updated
})

I wrote the three files above but didn't run or test any of them. A few choices change what you'd expect from the task wording:

- **Task 1:** signed-in users can only change their own profile. If the `id` in the URL isn't theirs, the request fails with 403. Only `name` and `email` are accepted. A request that also sends something like `role` is rejected rather than quietly ignoring the extra field. Because the schema doesn't make `email` unique, the handler checks for duplicates itself and returns 409 if another user has that email. The response never includes `passwordHash`.
- **Task 2:** I cached the stats function by user ID rather than caching the whole endpoint. Caching the endpoint as a whole risks showing one user's stats to another. The session check still runs on every request, and the response is marked `private, no-store` so browsers and proxies don't keep a copy.
- **Task 3:** `apiBase` sits under `runtimeConfig.public`, so both server and browser code can read it. Set the `NUXT_PUBLIC_API_BASE` environment variable to override it. Keep in mind that anything under `public` is visible to the browser.
