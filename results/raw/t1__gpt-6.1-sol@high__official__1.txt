=== FILE: app/components/DeleteUserModal.vue ===
<script setup lang="ts">
import { computed, ref, watch } from 'vue'

const props = defineProps<{
  open: boolean
  user: { id: string; name: string }
}>()

const emit = defineEmits<{
  'update:open': [open: boolean]
  confirm: []
}>()

const typedName = ref('')
const nameMatches = computed(() => typedName.value === props.user.name)

watch(
  () => [props.open, props.user.id, props.user.name],
  () => {
    typedName.value = ''
  },
  { flush: 'sync' }
)

function confirmDelete() {
  if (!props.open || !nameMatches.value) return

  emit('confirm')
  emit('update:open', false)
}
</script>

<template>
  <UModal
    :open="props.open"
    title="Delete user"
    description="Confirm deletion by typing the user's exact name."
    :ui="{ content: 'max-w-2xl', footer: 'justify-end' }"
    @update:open="emit('update:open', $event)"
  >
    <template #body>
      <div class="space-y-4">
        <p class="text-sm text-muted">
          Deleting <strong class="text-highlighted">{{ props.user.name }}</strong>
          is permanent and cannot be undone.
        </p>

        <UFormField label="Type the user's exact name to confirm">
          <UInput
            v-model="typedName"
            class="w-full"
            autocomplete="off"
            :spellcheck="false"
          />
        </UFormField>
      </div>
    </template>

    <template #footer>
      <UButton
        label="Cancel"
        color="neutral"
        variant="outline"
        @click="emit('update:open', false)"
      />
      <UButton
        label="Delete"
        color="error"
        :disabled="!nameMatches"
        @click="confirmDelete"
      />
    </template>
  </UModal>
</template>
=== END ===
