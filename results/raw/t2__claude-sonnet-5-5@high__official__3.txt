interface Order {

const PAGE_SIZE = 20

  { label: 'Newest first', value: 'createdAt' },
  { label: 'Customer', value: 'customer' },
  { label: 'Status', value: 'status' },
  { label: 'Total', value: 'total' }
]
const DEFAULT_SORT = sortOptions[0]!.value

useSeoMeta({ title: 'Orders' })

const route = useRoute()
const router = useRouter()

const page = computed<number>({
  get() {
    const value = Number(route.query.page)
    return Number.isInteger(value) && value > 0 ? value : 1
  },
  set(value) {
    router.push({ query: { ...route.query, page: String(value) } })
  }
})

const sort = computed<string>({
  get() {
    const value = route.query.sort
    return typeof value === 'string' && sortOptions.some(option => option.value === value)
      ? value
      : DEFAULT_SORT
  },
  set(value) {
    // Changing the sort always returns to the first page.
    router.push({ query: { ...route.query, sort: value, page: '1' } })
  }
})

const { data, status, error, refresh } = await useFetch<OrdersResponse>('/api/orders', {
  query: { page, sort },
  default: () => ({ items: [], total: 0 })
})

const currencyFormatter = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' })

const columns: TableColumn<Order>[] = [
  { accessorKey: 'id', header: 'Order' },

const INSTALLED = Symbol.for('app:api-auth-installed')

type TokenGetter = () => string | null | undefined

function isApiRequest(request: unknown): boolean {
  const base = import.meta.client ? window.location.origin : 'http://internal.invalid'

  let raw: string
  if (typeof request === 'string') {
    raw = request
  } else if (request instanceof URL) {
    raw = request.href
  } else if (request && typeof (request as Request).url === 'string') {
    raw = (request as Request).url
  } else {
    return false
  }

  try {
    const url = new URL(raw, base)
    return url.origin === base && (url.pathname === '/api' || url.pathname.startsWith('/api/'))
  } catch {
    return false
  }
}

function install(getToken: TokenGetter) {
  const original = globalThis.$fetch

  // Replace the global $fetch so $fetch, useFetch and useAsyncData all get the header.
  globalThis.$fetch = original.create({
    onRequest({ request, options }) {
      if (!isApiRequest(request)) return

      const token = getToken()
      if (!token) return

      const headers = new Headers(options.headers as HeadersInit | undefined)
      headers.set('Authorization', `Bearer ${token}`)
      options.headers = headers
    }
  })
}

export default defineNuxtPlugin({
  name: 'api-auth',
  setup() {
    const globals = globalThis as Record<symbol, unknown>

    // The global $fetch is shared by every request on the server, so install the wrapper
    // only once per process and resolve the token from the current request at call time.
    if (globals[INSTALLED]) return

    if (import.meta.client) {
      const token = useCookie<string | null>('token')
      install(() => token.value)
    } else {
      install(() => (tryUseNuxtApp() ? useCookie<string | null>('token').value : null))
    }

    globals[INSTALLED] = true
  }
})
